\begin{definition}{Vector space}{vs-test}
A \term{vector space} over a field $\K$ is a set $V$ with an addition and a
scalar multiplication satisfying the eight usual axioms.
\end{definition}

\begin{theorem}{Rank--nullity}{rn-test}
If $u\colon V\To W$ is linear and $\dim V<\infty$, then
$\dim V=\rank u+\dim\Ker u$.
\end{theorem}

\begin{proof}
Extend a basis of $\Ker u$ to a basis of $V$ and check that the images of the
added vectors form a basis of $\im u$.
\end{proof}

\begin{exercise}{Kernel of a projection}{proj}
Let $p$ be a projection. Show that $V=\Ker p\oplus\im p$.
\end{exercise}

\begin{problem}{University of Constantine, Analysis, 2016}{constantine-2016}
Let $A\in\Mat{n}{\R}$ with $A^{2}=A$. Determine $\spec(A)$ and show that $A$ is
diagonalisable.
\end{problem}

\begin{solution}{Exercise 1.1}
If $x\in\Ker p\cap\im p$ then $x=p(y)$ and $p(x)=0$, so $x=p(y)=p^{2}(y)=p(x)=0$.
Moreover $x=(x-p(x))+p(x)$ for every $x\in V$.
\end{solution}
