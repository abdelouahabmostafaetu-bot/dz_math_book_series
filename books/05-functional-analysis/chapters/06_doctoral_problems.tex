\section{Université Mohamed Khider --- Biskra}
\begin{problem}{(Biskra) — Épreuve N°02}{fa:biskra-2021-e2-p1}
Soit $H$ un espace de Hilbert et $F$ un sous-espace vectoriel fermé de $H$. On note $P_F:H\to H$ le projecteur orthogonal sur $F$, caractérisé par : pour tout $x\in H$, $P_F(x)\in F$ et $x-P_F(x)\in F^\perp$.
\begin{enumerate}\item Montrer que $F=\{x\in H : P_F(x)=x\}$.\item Montrer que $P_F$ est linéaire.\item Montrer que $P_F$ est continue et que $\|P_F\|=1$ (si $F\neq\{0\}$).\item Montrer que $\operatorname{Ker}(P_F)=F^\perp$.\end{enumerate}
\source{\emph{\small Université Mohamed Khider - Biskra, 2021, Épreuve~n\textsuperscript{o}~02.}}\end{problem}

\begin{problem}{(Biskra) — Épreuve N°02}{fa:biskra-2021-e2-p2}

\begin{enumerate}\item Montrer que $T$ est continue et calculer $\|T\|$.\item Déterminer $\operatorname{Ker}(T)$ et donner $g_0$ tel que $\operatorname{Ker}(T)=(\operatorname{Vect}\{g_0\})^\perp$.\item Calculer la projection orthogonale de $h(t)=t$ sur $\operatorname{Ker}(T)$.\end{enumerate}
\source{\emph{\small Université Mohamed Khider - Biskra, 2021, Épreuve~n\textsuperscript{o}~02.}}\end{problem}

\begin{problem}{(Guelma) — Épreuve N°01}{fa:guelma-2017-e1-p1}
Soit $E$ un espace vectoriel normé réel, $x_1,\dots,x_n\in E$ et $c_1,\dots,c_n\in\mathbb R$. Montrer qu'il existe une forme linéaire continue $f\in E'$ telle que $f(x_i)=c_i$ pour tout $i$ et $\|f\|\le\ell$ si et seulement si
$$\left|\sum_{i=1}^n\lambda_i c_i\right|\le\ell\left\|\sum_{i=1}^n\lambda_i x_i\right\|\quad\text{pour tous }\lambda_1,\dots,\lambda_n\in\mathbb R.$$
Soit $(T_n)$ une suite d'endomorphismes continus d'un espace de Banach $E$ telle que $T_nx\to T(x)$ pour tout $x\in E$. Montrer que $T$ est linéaire continu et que $\|T\|\le\liminf_n\|T_n\|$.
\source{\emph{\small Université 8 Mai 1945 - Guelma, 2017, Épreuve~n\textsuperscript{o}~01.}}\end{problem}

\begin{problem}{(Guelma) — Épreuve N°01}{fa:guelma-2014-e1-p1}
Soit $\Omega$ un ouvert borné de $\mathbb{R}^3$ à frontière de classe $C^1$. On pose
$$V^0=\left\{u\in L^2(\Omega,\mathbb{R}^3):\int_\Omega u\cdot\nabla\varphi\,dx=0,\ \forall\varphi\in H^1(\Omega)\right\},$$
et
$$V^1=\{u\in H^1(\Omega,\mathbb{R}^3):\operatorname{div}u=0,\ u|_{\partial\Omega}=0\}.$$
\begin{enumerate}\item Montrer que $V^0$ et $V^1$ sont des espaces vectoriels.\item Montrer que $V^1\subset V^0$.\item Montrer qu'il existe $c_1,c_2>0$ tels que, pour tout $u\in V^1$,
$$\|u\|_{H^1(\Omega)}^2\le c_1\int_\Omega\sum_{i,j=1}^3\left(\frac{\partial u_i}{\partial x_j}\right)^2dx\le c_2\|u\|_{H^1(\Omega)}^2.$$
\item Montrer que, pour $v\in V^1\cap H^2(\Omega)$ et $w\in V^1$,
$$(v,w)_{V^1}=(-\Delta v,w)_{L^2}=(-P_{V^0}\Delta v,w)_{L^2},$$
où $P_{V^0}$ est la projection orthogonale de $L^2(\Omega,\mathbb{R}^3)$ sur $V^0$.\item Pour $f\in V^0$, utiliser le théorème de Riesz afin de montrer que $-P_{V^0}\Delta v=f$ admet une unique solution faible $v\in V^1$.\end{enumerate}
\source{\emph{\small Université 8 Mai 1945 - Guelma, 2014, Épreuve~n\textsuperscript{o}~01.}}\end{problem}

\section{Université des Sciences et de la Technologie Houari Boumediène (USTHB)}
\begin{problem}{(USTHB) — Épreuve N°14}{fa:usthb-2016-e14-p1}
\begin{enumerate}\item Soit $\varphi$ une fonction numérique $\in\mathcal{C}^{\infty}(\mathbb{R})$. En utilisant la formule de Taylor, montrer que
$$\psi(x)=\begin{cases}\dfrac{\varphi(x)-\varphi(0)}{x} & \text{si } x\neq0,\\[2mm] \varphi'(0) & \text{si } x=0,\end{cases}$$
est dans $\mathcal{C}^{\infty}(\mathbb{R})$.
\item Pour $\varphi\in\mathcal{C}_0^{\infty}(\mathbb{R})$, on pose
$$\left\langle PV\frac{1}{x},\varphi\right\rangle:=\lim_{\varepsilon\downarrow0}\int_{|x|\geq\varepsilon}\frac{\varphi(x)}{x}\,dx.$$
\begin{enumerate}\item Soit $\varepsilon>0$. Montrer que $\left\langle PV\frac{1}{x},\varphi\right\rangle=\int_{-m}^{m}\psi(x)\,dx$, où $\psi$ est la fonction définie dans la première question et $m$ est tel que $\mathrm{supp}\,\varphi\subset[-m,+m]$.\item En déduire que $PV\dfrac{1}{x}$ est une distribution.\end{enumerate}
\item Soit $i$ le nombre imaginaire tel que $i^2=-1$. Montrer de même que $\dfrac{1}{x+i0}$ et $\dfrac{1}{x-i0}$ définies par
$$\left\langle\frac{1}{x+i0},\varphi\right\rangle:=\lim_{\varepsilon\downarrow0}\int_{\mathbb{R}}\frac{\varphi(x)}{x+i\varepsilon}\,dx,\qquad\left\langle\frac{1}{x-i0},\varphi\right\rangle:=\lim_{\varepsilon\downarrow0}\int_{\mathbb{R}}\frac{\varphi(x)}{x-i\varepsilon}\,dx$$
sont des distributions.\end{enumerate}
\source{\emph{\small Université des Sciences et de la Technologie Houari Boumediène (USTHB), 2016, Épreuve~n\textsuperscript{o}~14.}}\end{problem}

\begin{problem}{(USTHB) — Épreuve N°15}{fa:usthb-2016-e15-p1}
On s'intéresse à la recherche, par une méthode variationnelle, de la fonction $u$ vérifiant le problème aux limites $(P)$ suivant :
$$(P)\ \begin{cases}-\dfrac{d^2u}{dx^2}(x)=f(x), & x\in\,]0,1[,\\ u(0)=u(1)=0,\end{cases}$$
où $f$ est donnée dans $L^2(]0,1[)$.
\begin{enumerate}\item On cherche $u\in H_0^1(]0,1[)$ solution de $(P)$. Montrer que $u$ est solution du problème variationnel $(PV)$ : trouver $u\in H_0^1(]0,1[)$ tel que $a(u,v)=L(v)$ pour tout $v\in H_0^1(]0,1[)$. On explicitera, pour tout $(u,v)\in H_0^1\times H_0^1$, la forme bilinéaire $a(\cdot,\cdot)$ et la forme linéaire $L(\cdot)$.\item Montrer que la forme bilinéaire $a(\cdot,\cdot)$ est continue, symétrique sur $H_0^1\times H_0^1$, coercive sur $H_0^1$, et que la forme linéaire $L(\cdot)$ est continue sur $H_0^1(]0,1[)$.\item Montrer que l'application $v\mapsto\sqrt{a(v,v)}$ est une norme équivalente à la norme usuelle de $H^1(]0,1[)$; on notera $|||v|||=\sqrt{a(v,v)}$.\item Soit le produit scalaire défini par $(u,v)_1=a(u,v)$. Montrer que $(H_0^1(]0,1[),|||\cdot|||)$ est un espace de Hilbert.\item Rappeler le théorème de représentation de Riesz. En déduire l'existence et l'unicité (sans faire appel au théorème de Lax-Milgram) de la solution du problème variationnel.\end{enumerate}
\source{\emph{\small Université des Sciences et de la Technologie Houari Boumediène (USTHB), 2016, Épreuve~n\textsuperscript{o}~15.}}\end{problem}

\begin{problem}{(USTHB) — Épreuve N°15}{fa:usthb-2016-e15-p2}
Soient $\Omega$ un ouvert borné régulier de $\mathbb{R}^2$ de bord $\Gamma$ et $f\in L^{\infty}(\mathbb{R})$, $f\geq0$ p.p. On suppose que $f$ est lipschitzienne. On considère le problème non linéaire suivant :
$$\begin{cases}-\Delta u+u^2=f(u) & \text{dans } \Omega,\\ u=0 & \text{sur } \Gamma.\end{cases}$$
La solution faible est $u\in H_0^1(\Omega)$ telle que $u\geq0$ p.p. et
$$\int_{\Omega}\nabla u\cdot\nabla v+u^2v\,dx=\int_{\Omega}f(u)v\,dx,\quad\forall v\in H_0^1(\Omega).$$
\begin{enumerate}\item Montrer qu'on peut définir par récurrence une suite de fonctions $(u_n)$ telle que $u_0=0$ et, pour $n\geq1$, $u_n$ est la solution du problème linéaire
$$\begin{cases}-\Delta u_n+u_{n-1}u_n=f(u_{n-1}) & \text{dans } \Omega,\\ u_n=0 & \text{sur } \Gamma,\end{cases}$$
de sorte que $u_n\in H_0^1(\Omega)$, $u_n\geq0$ p.p. et qu'il existe une constante $C>0$ indépendante de $n$ telle que $\|u_n\|_{H^1(\Omega)}\leq C$.\item En déduire que la suite $(u_n)$ admet une sous-suite qui converge fortement dans $L^2(\Omega)$ vers une limite $u\in H^1(\Omega)$.\end{enumerate}
\source{\emph{\small Université des Sciences et de la Technologie Houari Boumediène (USTHB), 2016, Épreuve~n\textsuperscript{o}~15.}}\end{problem}

\begin{problem}{(Batna 1) — Épreuve N°01}{fa:batna1-2015-e1-p1}
Considérons le système suivant sur $L^2(0,1)$ :
$$\begin{cases}\dfrac{\partial y}{\partial t}(x,t)=\dfrac{\partial^2y}{\partial x^2}(x,t), & 0<x<1,\ t>0,\\[1ex] y(0,t)=y(1,t)=0, & t>0,\\[1ex] z(t)=\displaystyle\int_0^1c(x)y(x,t)\,dx, & t>0,\end{cases}$$
où $c(x)=\mathbf{1}_{[1-\varepsilon,\,1]}(x)$ et $\varepsilon>0$ est tel que $0<1-\varepsilon<1$.
\begin{enumerate}\item Réécrire ce système sous la forme $y'(t)=Ay(t)$, $z(t)=Cy(t)$.\item Préciser l'espace d'état $Y$ et l'espace de sorties $Z$.\item Montrer que $A$ est un opérateur spectral de Riesz. En déduire que $A$ engendre un $C_0$-semi-groupe $S(t)$ sur $Y$. Donner l'expression de ce semi-groupe.\item Montrer que $C\in\mathcal{L}(Y,Z)$.\item Étudier l'observabilité approchée du système.\end{enumerate}
\source{\emph{\small Université Batna 1 - Hadj Lakhdar, 2015, Épreuve~n\textsuperscript{o}~01.}}\end{problem}

\begin{problem}{(Batna 1) — Épreuve N°01}{fa:batna1-2013-e1-p1}
Soit $c>0$. On considère le problème de Cauchy suivant
$$\begin{cases}\partial_t^2u(t,x)-c^2\Delta_xu(t,x)=0 & \text{si } (t,x)\in\,]0,\infty[\times\mathbb{R}^d,\\ u(0,x)=f(x),\quad\partial_tu(0,x)=g(x) & \text{si } x\in\mathbb{R}^d.\end{cases}\tag{1}$$
Supposons que $f,g\in\mathcal{S}(\mathbb{R}^d,\mathbb{C})$.
\begin{enumerate}\item En utilisant la transformée de Fourier partielle par rapport à $x$, prouver que le problème (1) admet une unique solution $u$, donnée par
$$\hat{u}(t,\xi)=\hat{f}(\xi)\cos(c|\xi|t)+\hat{g}(\xi)\,\frac{\sin(c|\xi|t)}{c|\xi|},\qquad t\in[0,+\infty[,\ \xi\in\mathbb{R}^d\setminus\{0\}.\tag{2}$$
\item Déduire que $u\in C^\infty([0,\infty[\times\mathbb{R}^d)$.\item Prouver que pour tout $\xi\in\mathbb{R}^d$, la solution (2) vérifie
$$(\Re(\partial_t\hat{u}))^2+c^2|\xi|^2(\Re(\hat{u}))^2=(\Re(\hat{g}))^2+c^2|\xi|^2(\Re(\hat{f}))^2,$$
$$(\Im(\partial_t\hat{u}))^2+c^2|\xi|^2(\Im(\hat{u}))^2=(\Im(\hat{g}))^2+c^2|\xi|^2(\Im(\hat{f}))^2.$$
\item Soit $s\in\mathbb{R}$. Prouver que pour tout $t\in[0,\infty[$ :
$$\|\partial_tu(t)\|_{H^s}^2+c^2\|\nabla_xu(t)\|_{H^s}^2\leq\|g\|_{H^s}^2+c^2\|\nabla_xf\|_{H^s}^2.$$\end{enumerate}
Notations : $\Delta_x=\sum_{i=1}^{d}\frac{\partial^2}{\partial x_i^2}$ ; $\Re$ : partie réelle ; $\Im$ : partie imaginaire ; $\mathcal{F}$ : la transformée de Fourier ; $|\cdot|$ : la norme euclidienne.
\source{\emph{\small Université Batna 1 - Hadj Lakhdar, 2013, Épreuve~n\textsuperscript{o}~01.}}\end{problem}

\begin{problem}{(Batna 1) — Épreuve N°01}{fa:batna1-2013-e1-p2}
\begin{enumerate}\item Soit $u\in L^\infty(\mathbb{R})$ et continue en $0$. Prouver que
$$\lim_{n\to\infty}\int_{\mathbb{R}}u(x)\left(\frac{n}{2\pi}\right)^{1/2}e^{-\frac{nx^2}{2}}\,dx=u(0).$$
Indication : supposer que $u(0)=0$ et utiliser la continuité de $u$ en $0$ pour découper l'intégrale en deux, l'une portant sur un voisinage de $0$, et l'autre sur l'extérieur de ce voisinage.
\item Soit $\xi\in\mathbb{R}$, on note $(1+|\xi|^2)^{1/2}$ par $\langle\xi\rangle$. On considère une fonction $f\in L^2(\mathbb{R})$, $f\geq0$, telle que $\langle\xi\rangle^{-1}f\notin L^1(\mathbb{R})$. On pose
$$u=\mathcal{F}^{-1}\left(\langle\xi\rangle^{-1}f\right),$$
où $\mathcal{F}$ désigne l'opérateur de Fourier au sens des distributions tempérées.
\begin{enumerate}\item Prouver que $u\in H^{1/2}(\mathbb{R})$.\item Définissons la suite $(I_n)_{n\in\mathbb{N}}$ par
$$I_n=\int_{\mathbb{R}}u(x)\left(\frac{n}{2\pi}\right)^{1/2}e^{-\frac{nx^2}{2}}\,dx.$$
Prouver que pour tout $n\in\mathbb{N}$, $I_n=\int_{\mathbb{R}}\hat{u}(\xi)\,e^{-\frac{\xi^2}{2n}}\,d\xi$, puis, en utilisant le lemme de Fatou, prouver que $\lim_{n\to\infty}I_n=+\infty$.\end{enumerate}
\item En déduire que $H^{1/2}(\mathbb{R})$ n'est pas contenu dans $L^\infty(\mathbb{R})$.\end{enumerate}
\source{\emph{\small Université Batna 1 - Hadj Lakhdar, 2013, Épreuve~n\textsuperscript{o}~01.}}\end{problem}

\begin{problem}{(M'Sila) — Épreuve N°10}{fa:msila-2015-e10-p1}
On considère le problème d'évolution suivant :
$$\begin{cases}\partial_tu-i\,\partial_{xx}u=0, & x\in\mathbb{R},\ t>0,\\ u(x,0)=\varphi(x), & \varphi\in\mathcal{S}.\end{cases}$$
\begin{enumerate}\item Appliquer la transformée de Fourier pour résoudre le problème.\item Démontrer que $\lVert u(t)\rVert_{L^\infty(\mathbb{R})}\leq\frac{1}{\sqrt{4\pi t}}\lVert\varphi\rVert_{L^1(\mathbb{R})}$.\item Utiliser l'identité de Parseval pour démontrer que $\lVert u(t)\rVert_{L^2(\mathbb{R})}\leq\lVert\varphi\rVert_{L^2(\mathbb{R})}$.\end{enumerate}
\source{\emph{\small Université Mohamed Boudiaf - M'Sila, 2015, Épreuve~n\textsuperscript{o}~10.}}\end{problem}
