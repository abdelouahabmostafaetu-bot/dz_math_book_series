\section{Université des Sciences et de la Technologie Houari Boumediène (USTHB)}
\begin{problem}{(USTHB) — Épreuve N°02}{top:usthb-2026-e2-p1}
Let $\mathcal T_u$ be the usual topology on $\mathbb R$ and let $\mathcal T=\{\varnothing,\mathbb R\}\cup\{U:U^c\text{ is compact in the usual topology}\}$. Prove that $\mathcal T$ is a topology, determine whether it is Hausdorff and separable, and prove compactness.
\source{\emph{\small Université des Sciences et de la Technologie Houari Boumediène (USTHB), 2026, Épreuve~n\textsuperscript{o}~02.}}\end{problem}

\section{Université Kasdi Merbah --- Ouargla}
\begin{problem}{(Ouargla) — Épreuve N°01}{top:ouargla-2025-e1-p2}
For a metric space $(E,d)$ prove that $\rho(x,y)=d(x,y)/(3+2d(x,y))$ is a distance, determine $B_\rho(a,1)$, compare balls, and compare the topologies.
\source{\emph{\small Université Kasdi Merbah - Ouargla, 2025, Épreuve~n\textsuperscript{o}~01.}}\end{problem}

\section{Université Echahid Hamma Lakhdar --- El Oued}
\begin{problem}{(El Oued) — Épreuve N°01}{top:eloued-2025-e1-p2}
Let $(X,d)$ be compact and $f:X\to X$ satisfy $x\ne y\Rightarrow d(f(x),f(y))<d(x,y)$. Prove that $g(x)=d(x,f(x))$ is continuous and that its minimum is attained at a unique fixed point.
\source{\emph{\small Université Echahid Hamma Lakhdar - El Oued, 2025, Épreuve~n\textsuperscript{o}~01.}}\end{problem}

\section{Université Amar Telidji --- Laghouat}
\begin{problem}{(Laghouat) — Épreuve N°01}{top:laghouat-2025-e1-p2}
For $E=\mathbb R$ prove that $d(x,y)=|\arctan x-\arctan y|$ is a distance, calculate the ball of radius $\pi/4$, and compare convergence with the discrete metric.
\source{\emph{\small Université Amar Telidji - Laghouat, 2025, Épreuve~n\textsuperscript{o}~01.}}\end{problem}
