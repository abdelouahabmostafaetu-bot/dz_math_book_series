\section{Université des Sciences et de la Technologie Houari Boumediène (USTHB)}
\begin{problem}{(USTHB) — Épreuve N°01}{pde:usthb-2026-e1-p1}
Soit $\Omega\subset\mathbb R^2$ un ouvert borné de frontière régulière. On considère le problème variationnel
$$\int_\Omega\nabla u\cdot\nabla v\,dxdy+\int_\Omega(\nabla u\cdot F)v\,dxdy=\int_\Omega f\cdot\nabla v\,dxdy,\qquad\forall v\in H_0^1(\Omega),$$
avec $f\in(L^2(\Omega))^2$, $F\in(L^\infty(\Omega))^2$ et $\operatorname{div}(F)=0$. Montrer que $\nabla u\cdot F=\operatorname{div}(uF)$, que $\int_\Omega u(F\cdot\nabla u)=0$, établir l'existence et l'unicité de $u\in H_0^1(\Omega)$, puis déterminer le problème aux limites satisfait par $u$ lorsque $u\in H^2(\Omega)$ et $\operatorname{div}(f)\in L^2(\Omega)$.
\source{\emph{\small Université des Sciences et de la Technologie Houari Boumediène (USTHB), 2026, Épreuve~n\textsuperscript{o}~01.}}\end{problem}

\begin{problem}{(M'Sila) — Épreuve N°04}{pde:msila-2016-e4-p1}
Soit $v(x,t)$ la température au temps $t$ au point $x$, avec $-L\leq x\leq L$, et $\varphi\in L^2(-L,L)$. On suppose qu'elle satisfait l'équation de la chaleur
$$\begin{cases}\partial_tv-\partial_{xx}v=0, & -L<x<L,\ t>0,\\ v(-L,t)=v(L,t)=0, & t>0,\\ v(x,0)=\varphi(x), & -L<x<L.\end{cases}$$
\begin{enumerate}\item Démontrer que la fonction $E(t)=\frac12\int_{-L}^{L}v^2(x,t)\,dx$ est une fonction décroissante en $t$.\item Soient $f$ et $g$ des fonctions données. Utiliser le résultat précédent pour démontrer l'unicité de la solution du problème suivant :
$$\begin{cases}\partial_tw-\partial_{xx}w=0, & -L<x<L,\ t>0,\\ w(-L,t)=f(t),\quad w(L,t)=g(t), & t>0,\\ w(x,0)=\psi(x), & -L<x<L.\end{cases}$$
\item Déduire que si $\psi(x)$ est une fonction impaire et $w(-L,t)=w(L,t)$, alors la température $w(x,t)$ est aussi une fonction impaire en $x$.\end{enumerate}
\source{\emph{\small Université Mohamed Boudiaf - M'Sila, 2016, Épreuve~n\textsuperscript{o}~04.}}\end{problem}

\begin{problem}{(M'Sila) — Épreuve N°04}{pde:msila-2016-e4-p2}
Soit le problème de Cauchy pour l'équation de la chaleur :
$$\begin{cases}\partial_tu=c\,\partial_{xx}u, & x\in\mathbb{R},\ t>0,\ c>0,\\ u(x,0)=\varphi(x), & x\in\mathbb{R}.\end{cases}$$
La solution du problème s'écrit comme $u(x,t)=\int_{\mathbb{R}}\varphi(y)K(x,y,t)\,dy$, où
$$K(x,y,t)=\frac{1}{\sqrt{4\pi ct}}e^{-\frac{(y-x)^2}{4ct}}$$
est le noyau de Gauss, solution fondamentale.
\begin{enumerate}\item Montrer que $K(x,y,t)>0$ et $K(x,y,t)\in C^\infty$.\item Montrer que $K(x,y,t)$ est également solution de l'équation, c'est-à-dire $\partial_tK=c\,\partial_{xx}K$.\item Des questions précédentes, que peut-on déduire pour la solution $u(x,t)$ ?\item Montrer que $\int_{\mathbb{R}}K(x,y,t)\,dy=1$.\item Déduire que si $0<\varphi(x)<M$, alors $0<u(x,t)<M$, pour une constante positive $M$.\item Calculer la solution pour
$$\varphi(x)=\begin{cases}1, & \lvert x\rvert<1,\\ 0, & \lvert x\rvert\geq1,\end{cases}$$
sachant que $\operatorname{Erf}(x)=\frac{2}{\sqrt{\pi}}\int_0^xe^{-s^2}\,ds$. Indication : $\int_{\mathbb{R}}e^{-u^2}\,du=\sqrt{\pi}$.\end{enumerate}
\source{\emph{\small Université Mohamed Boudiaf - M'Sila, 2016, Épreuve~n\textsuperscript{o}~04.}}\end{problem}

\begin{problem}{(M'Sila) — Épreuve N°01}{pde:msila-2015-e1-p1}
On considère le problème d'évolution suivant :
$$\begin{cases}\partial_tu-\partial_{xx}u+f(u)=0, & (t,x)\in\,]0,+\infty[\times]0,1[,\\ u(t,0)=u(t,1)=0, & \forall t>0,\\ u(0,x)=g(x), & \forall x\in\,]0,1[,\end{cases}$$
où $f:\mathbb{R}\to\mathbb{R}$ est une fonction continue telle que $f(u)\cdot u\geq0$ et $g\in L^2(0,1)$. On note $P(t)=\lVert u(t)\rVert_{L^2(0,1)}^2$ et on veut obtenir des estimations de plus en plus fines pour $P(t)$, sans résoudre l'EDP.
\begin{enumerate}\item Démontrer que $P'(t)\leq0$ et déduire que $P(t)\leq\lVert g\rVert_{L^2(0,1)}^2$ pour tout $t>0$.\item En utilisant l'inégalité de Poincaré, démontrer que $P'(t)\leq-2\pi^2P(t)$ et déduire que $P(t)\leq P(0)e^{-2\pi^2t}$.\item On prend maintenant $f(u)=u^3$. En utilisant l'inégalité de Cauchy-Schwarz pour estimer $\int_0^1u^4(t)\,dx$, démontrer que $P'(t)\leq-2\pi^2P(t)-2P^2(t)$ et déduire que
$$P(t)\leq\left(\frac{\pi^2P(0)}{\pi^2+P(0)\left(1-e^{-2\pi^2t}\right)}\right)e^{-2\pi^2t}.$$\end{enumerate}
Aide : la constante de Poincaré est $\frac1\pi$. Le changement de variable $Z(t)=\frac1{P(t)}$ peut être utile dans l'inégalité précédente.
\source{\emph{\small Université Mohamed Boudiaf - M'Sila, 2015, Épreuve~n\textsuperscript{o}~01.}}\end{problem}

\section{Université Badji Mokhtar --- Annaba}
\begin{problem}{(Annaba) — Épreuve N°01}{pde:annaba-2015-e1-p1}
Soit $\Omega\subset\mathbb{R}^n$ un ouvert borné avec frontière $\partial\Omega$ assez régulière et $f\in L^2(\Omega)$. On dit que $u\in H^1(\Omega)$ est solution faible du problème de Neumann
$$\begin{cases}-\Delta u=f & \text{sur } \Omega,\\ \dfrac{\partial u}{\partial\nu}=0 & \text{sur } \partial\Omega,\end{cases}$$
si
$$\int_\Omega\nabla u\cdot\nabla v\,dx=\int_\Omega f(x)v(x)\,dx,\quad\forall v\in H^1(\Omega).$$
\begin{enumerate}\item Montrer que si $u\in H^2(\Omega)$ est solution forte alors la formulation faible est vérifiée.\item Si $u$ est solution faible et $u\in H^2(\Omega)$, montrer que : (a) $-\Delta u=f$ dans $L^2(\Omega)$ ; (b) $\partial u/\partial\nu=0$ dans $L^2(\partial\Omega)$ au sens des traces.\item Montrer que la condition $\int_\Omega f(x)\,dx=0$ est nécessaire pour l'existence d'une solution faible.\item Soit $H=\{v\in H^1(\Omega) : \int_\Omega v(x)\,dx=0\}$. En supposant la condition précédente satisfaite, montrer que si $u\in H$ vérifie la formulation faible pour tout $v\in H$, alors elle est vérifiée pour tout $v\in H^1(\Omega)$.\item Utiliser Lax-Milgram dans l'espace de Hilbert $H$ et déduire que la condition de compatibilité est suffisante pour l'existence d'une solution faible.\end{enumerate}
\source{\emph{\small Université Badji Mokhtar - Annaba, 2015, Épreuve~n\textsuperscript{o}~01.}}\end{problem}
