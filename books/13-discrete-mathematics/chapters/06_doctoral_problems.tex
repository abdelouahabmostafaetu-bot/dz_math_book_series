\section{École Nationale Supérieure de Mathématiques (ENSM)}
\begin{problem}{(ENSM) — Épreuve générale}{dm:ensm-2026-general-p3}
Let $A$ be a finite field and equip $A^{\mathbb Z}$ with the metric defined by the first differing coordinate. A convolutional encoder is given by $(T(x))_m=\Phi(x_{m+L},\ldots,x_{m+R})$. Prove that every convolutional encoder is continuous and commutes with the shift, and determine whether the converse holds.
\source{\emph{\small École Nationale Supérieure de Mathématiques (ENSM), 2026, Épreuve générale.}}\end{problem}
