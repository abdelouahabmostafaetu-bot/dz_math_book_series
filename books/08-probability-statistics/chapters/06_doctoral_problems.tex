\section{École Nationale Supérieure de Mathématiques (ENSM)}
\begin{problem}{(ENSM) — Épreuve générale}{ps:ensm-2026-general-p2}
For $k$ urns containing balls numbered $1$ to $n$, let $X_n$ be the maximum of the $k$ numbers drawn. Determine $\Omega$, the law of $X_n$, and an equivalent of $\mathbb E(X_n)$ as $n\to\infty$ with $k$ fixed.
\source{\emph{\small École Nationale Supérieure de Mathématiques (ENSM), 2026, Épreuve générale.}}\end{problem}

\section{Université des Sciences et de la Technologie Houari Boumediène (USTHB)}
\begin{problem}{(USTHB) — Épreuve N°01}{ps:usthb-2026-e1-p1}
For the maximum $X_n$ of $k$ independent uniform draws from $\{1,\ldots,n\}$, calculate its distribution and determine an equivalent of $\mathbb E(X_n)$.
\source{\emph{\small Université des Sciences et de la Technologie Houari Boumediène (USTHB), 2026, Épreuve~n\textsuperscript{o}~01.}}\end{problem}

\section{Université Djilali Liabès --- Sidi Bel Abbès}
\begin{problem}{(Sidi Bel Abbès) — Épreuve N°01}{ps:sba-2021-e1-p2}
Pour un candidat répondant au hasard à dix questions, chacune offrant quatre affirmations dont une seule est exacte, déterminer la loi de $X$, son espérance, sa variance et $P(X\ge8)$.
\source{\emph{\small Université Djilali Liabès - Sidi Bel Abbès, 2021, Épreuve~n\textsuperscript{o}~01.}}\end{problem}
